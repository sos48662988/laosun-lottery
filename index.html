let rows = []; let winList = []; let head = []; let wonUIDs = []; const PROXY_PREFIX = 'https://images.weserv.nl/?url='; const DEFAULT_AVATAR = 'https://picsum.photos/50/50'; function calcValidCount() { if (rows.length === 0) return 0; const at = document.querySelector('input[name="at"]:checked').value; const dedupe = document.querySelector('input[name="repeat"]:checked').value; const blocks = document.getElementById('block').value.split('\n').map(i => i.trim().toLowerCase()).filter(Boolean); const st = document.getElementById('startTime').value ? new Date(document.getElementById('startTime').value) : null; const et = document.getElementById('endTime').value ? new Date(document.getElementById('endTime').value) : null; let pool = [...rows]; // 用户ID在D列：row[3]，全局UID去重 if (dedupe === 'dedupe') { const u = new Set(); pool = pool.filter(row => { const uid = String(row[3] ?? new Date(document.getElementById('startTime').value) : null; const et = document.getElementById('endTime').value ? new Date(document.getElementById('endTime').value) : null; let pool = [...rows]; if (dedupe === 'dedupe') { const u = new Set(); pool = pool.filter(row => { const uid = String(row[3] ?? '').trim(); if (!uid || u.has(uid)) return false; u.add(uid); return true; }) } pool = pool.filter(r => { const t = r.join('').toLowerCase(); if (at === 'forbid' && t.includes('@')) return false; if (blocks.some(b => t.includes(b))) return false; const ct = r[8] ? new Date(r[8]) : null; if (ct && st && ct < st) return false; if (ct && et && ct > et) return false; const uid = String(r[3] ?? '').trim(); if (uid && wonUIDs.includes(uid)) return false; return true; }); pool.sort(() => Math.random() - 0.5); winList = pool.slice(0, num); const newWins = winList.map(w => String(w[3] ?? '').trim()).filter(Boolean); wonUIDs = [...new Set([...wonUIDs, ...newWins])]; const res = document.getElementById('result'); res.innerHTML = ''; winList.forEach((w, i) => { const uid = w[3] || ''; const name = w[4] || '无'; const comment = w[7] || ''; const rawAvatar = w[14] || ''; const card = document.createElement('div'); card.className = 'winner-card'; const avatarContainer = document.createElement('div'); avatarContainer.className = 'avatar-container'; const img = document.createElement('img'); img.className = 'winner-avatar'; if (rawAvatar) { img.src = PROXY_PREFIX + encodeURIComponent(rawAvatar.trim()); } else { img.src = DEFAULT_AVATAR; } img.onerror = function() { this.src = DEFAULT_AVATAR; }; const infoWrap = document.createElement('div'); infoWrap.className = 'winner-info-wrap'; infoWrap.innerHTML = `

🎊 第${i+1}名

UID：${uid}
暱稱：${name}
${comment}
`; avatarContainer.appendChild(img); card.appendChild(avatarContainer); card.appendChild(infoWrap); res.appendChild(card); }); updateStats(); }; document.getElementById('export').onclick = () => { if (!winList.length) { alert('请先抽奖'); return; } const ws = XLSX.utils.aoa_to_sheet([head, ...winList]); const wb = XLSX.utils.book_new(); XLSX.utils.book_append_sheet(wb, ws, '中奖名单'); XLSX.writeFile(wb, '老孫抽奖站_中奖名单.xlsx'); }; document.getElementById('clearRecord').onclick = () => { if (confirm('确定清空本次已抽记录？')) { wonUIDs = []; updateStats(); alert('已清空'); } };
